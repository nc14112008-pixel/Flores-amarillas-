  <script>
    function comenzarExperiencia() {
      const screen = document.getElementById('welcome-screen');
      screen.classList.add('hidden');
    }

    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();

    // Fondo espacial estilo galaxia
    function createGalaxyBackground() {
      const canvas = document.createElement('canvas');
      canvas.width = 1024;
      canvas.height = 1024;
      const ctx = canvas.getContext('2d');

      const grad = ctx.createRadialGradient(512, 512, 0, 512, 512, 700);
      grad.addColorStop(0, '#21021d');
      grad.addColorStop(0.5, '#0d0117');
      grad.addColorStop(1, '#020005');
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, 1024, 1024);

      for (let i = 0; i < 8; i++) {
        const x = Math.random() * 1024;
        const y = Math.random() * 1024;
        const rad = 150 + Math.random() * 250;
        const nebGrad = ctx.createRadialGradient(x, y, 0, x, y, rad);
        nebGrad.addColorStop(0, 'rgba(230, 0, 120, 0.25)');
        nebGrad.addColorStop(0.5, 'rgba(120, 0, 180, 0.12)');
        nebGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
        ctx.fillStyle = nebGrad;
        ctx.beginPath();
        ctx.arc(x, y, rad, 0, Math.PI * 2);
        ctx.fill();
      }

      for (let i = 0; i < 400; i++) {
        const x = Math.random() * 1024;
        const y = Math.random() * 1024;
        const size = Math.random() * 2;
        ctx.fillStyle = Math.random() > 0.3 ? '#ffffff' : '#ffd700';
        ctx.fillRect(x, y, size, size);
      }

      return new THREE.CanvasTexture(canvas);
    }

    scene.background = createGalaxyBackground();

    const camera = new THREE.PerspectiveCamera(55, window.innerWidth / window.innerHeight, 0.1, 1000);
    camera.position.set(0, 28, 12);
    camera.lookAt(0, 0, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.4;
    container.appendChild(renderer.domElement);

    // Iluminación
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
    scene.add(ambientLight);

    const centerGlow = new THREE.PointLight(0xffea00, 12, 45);
    centerGlow.position.set(0, 0, 0);
    scene.add(centerGlow);

    // Creador de Girasoles 3D
    function createSunflowerMesh() {
      const group = new THREE.Group();

      const centerGeo = new THREE.CylinderGeometry(0.7, 0.7, 0.25, 32);
      const centerMat = new THREE.MeshStandardMaterial({ color: 0x2b1502, roughness: 0.8 });
      const center = new THREE.Mesh(centerGeo, centerMat);
      center.rotation.x = Math.PI / 2;
      group.add(center);

      const petalGeo = new THREE.ConeGeometry(0.22, 1.2, 8);
      const petalMat = new THREE.MeshStandardMaterial({ color: 0xffc800, roughness: 0.3 });

      const petalCount = 20;
      for (let i = 0; i < petalCount; i++) {
        const angle = (i / petalCount) * Math.PI * 2;
        const petal = new THREE.Mesh(petalGeo, petalMat);
        petal.position.x = Math.cos(angle) * 1.0;
        petal.position.y = Math.sin(angle) * 1.0;
        petal.rotation.z = angle - Math.PI / 2;
        group.add(petal);
      }

      return group;
    }

    // --- ARO CENTRAL FORMADO COMPLETAMENTE POR FLORES ---
    const centerRingGroup = new THREE.Group();

    // Núcleo brillante
    const coreGeo = new THREE.SphereGeometry(1.5, 32, 32);
    const coreMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
    const coreMesh = new THREE.Mesh(coreGeo, coreMat);
    centerRingGroup.add(coreMesh);

    // Halo Neón
    function createGlowTexture() {
      const canvas = document.createElement('canvas');
      canvas.width = 256; canvas.height = 256;
      const ctx = canvas.getContext('2d');
      const gradient = ctx.createRadialGradient(128, 128, 0, 128, 128, 128);
      gradient.addColorStop(0, 'rgba(255, 255, 255, 1)');
      gradient.addColorStop(0.25, 'rgba(255, 230, 0, 0.95)');
      gradient.addColorStop(0.6, 'rgba(255, 120, 0, 0.4)');
      gradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
      ctx.fillStyle = gradient;
      ctx.fillRect(0, 0, 256, 256);
      return new THREE.CanvasTexture(canvas);
    }

    const glowSprite = new THREE.Sprite(new THREE.SpriteMaterial({
      map: createGlowTexture(),
      color: 0xffea00,
      transparent: true,
      blending: THREE.AdditiveBlending
    }));
    glowSprite.scale.set(13, 13, 1);
    centerRingGroup.add(glowSprite);

    // Corona de girasoles central
    const centerSunflowers = [];
    const ringFlowerCount = 18;
    const ringRadius = 3.6;

    for (let i = 0; i < ringFlowerCount; i++) {
      const angle = (i / ringFlowerCount) * Math.PI * 2;
      const sunflower = createSunflowerMesh();
      
      const x = Math.cos(angle) * ringRadius;
      const z = Math.sin(angle) * ringRadius;
      
      sunflower.position.set(x, 0, z);
      sunflower.scale.set(0.75, 0.75, 0.75);
      
      centerRingGroup.add(sunflower);
      centerSunflowers.push(sunflower);
    }

    scene.add(centerRingGroup);

    // --- ÓRBITAS CIRCULARES CONCÉNTRICAS ---
    const orbitRadii = [6, 9, 12, 15, 18];
    orbitRadii.forEach(r => {
      const orbitGeo = new THREE.RingGeometry(r - 0.03, r + 0.03, 128);
      const orbitMat = new THREE.MeshBasicMaterial({
        color: 0xffaaef,
        side: THREE.DoubleSide,
        transparent: true,
        opacity: 0.45
      });
      const orbitMesh = new THREE.Mesh(orbitGeo, orbitMat);
      orbitMesh.rotation.x = Math.PI / 2;
      scene.add(orbitMesh);
    });

    // Frases cariñosas y bonitas
    const lovePhrases = [
      "Te quiero muchísimo ❤️",
      "Cada día a tu lado es bonito ✨",
      "Gracias por existir 💖",
      "Iluminas todo con tu luz ☀️",
      "Un detalle para tu sonrisa 🌻",
      "Sos una persona muy especial 🌸",
      "Tu felicidad alegra mi día 😊",
      "Abrazos gigantes para vos 🤗",
      "Sos amor puro 💕",
      "Siempre radiante 🌟",
      "Amo verte feliz ✨",
      "Flores amarillas para vos 🌼",
      "Nunca dejes de brillar ❤️",
      "Un regalito del corazón 🎁",
      "Tu bondad es única 💖"
    ];

    const textColors = ["#ffffff", "#ffee55", "#66e0ff", "#ff88cc", "#ffaa00", "#d47aff"];

    function createTextTexture(text, colorHex) {
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');
      canvas.width = 1024;
      canvas.height = 256;

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      ctx.font = "Bold 48px sans-serif";
      ctx.fillStyle = colorHex;
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.shadowColor = colorHex;
      ctx.shadowBlur = 18;
      ctx.fillText(text, 512, 128);

      const texture = new THREE.CanvasTexture(canvas);
      texture.needsUpdate = true;
      return texture;
    }

    // Distribución en espiral de flores y frases
    const elementsGroup = new THREE.Group();
    const sunflowersList = [];
    const totalItems = 56;

    for (let i = 0; i < totalItems; i++) {
      const radius = 5.5 + (i / totalItems) * 14.5;
      const angle = (i * 0.45);
      const x = Math.cos(angle) * radius;
      const z = Math.sin(angle) * radius;
      const y = (Math.random() - 0.5) * 1.5;

      if (i % 2 === 0) {
        // Crear grupo contenedor de la flor + frase
        const flowerContainer = new THREE.Group();
        flowerContainer.position.set(x, y, z);

        const sunflower = createSunflowerMesh();
        const scale = 0.5 + Math.random() * 0.35;
        sunflower.scale.set(scale, scale, scale);
        flowerContainer.add(sunflower);

        // Añadir una frase cariñosa sobre este girasol
        const phrase = lovePhrases[Math.floor(i / 2) % lovePhrases.length];
        const color = textColors[Math.floor(Math.random() * textColors.length)];
        const texture = createTextTexture(phrase, color);

        const mat = new THREE.SpriteMaterial({ map: texture, transparent: true });
        const textSprite = new THREE.Sprite(mat);
        textSprite.position.set(0, 1.3, 0); // Posicionada justo sobre el girasol
        textSprite.scale.set(5.5, 1.4, 1);
        flowerContainer.add(textSprite);

        elementsGroup.add(flowerContainer);
        sunflowersList.push(sunflower);
      } else {
        // Sprites con frases independientes en la órbita
        const phrase = lovePhrases[Math.floor(i / 2) % lovePhrases.length];
        const color = textColors[Math.floor(Math.random() * textColors.length)];
        const texture = createTextTexture(phrase, color);

        const mat = new THREE.SpriteMaterial({ map: texture, transparent: true });
        const sprite = new THREE.Sprite(mat);

        sprite.position.set(x, y, z);
        sprite.scale.set(6.5, 1.6, 1);
        elementsGroup.add(sprite);
      }
    }

    scene.add(elementsGroup);

    // Destellos / Polvo de estrellas
    const particlesGeo = new THREE.BufferGeometry();
    const particlesCount = 300;
    const posArray = new Float32Array(particlesCount * 3);

    for (let i = 0; i < particlesCount * 3; i++) {
      posArray[i] = (Math.random() - 0.5) * 50;
    }

    particlesGeo.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
    const particlesMat = new THREE.PointsMaterial({
      size: 0.18,
      color: 0xffffff,
      transparent: true,
      opacity: 0.9
    });

    const particlesMesh = new THREE.Points(particlesGeo, particlesMat);
    scene.add(particlesMesh);

    // Animación
    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const elapsedTime = clock.getElapsedTime();

      elementsGroup.rotation.y = elapsedTime * 0.06;
      particlesMesh.rotation.y = -elapsedTime * 0.02;
      centerRingGroup.rotation.y = elapsedTime * 0.1;

      // Pulsación de luz central
      const pulse = Math.sin(elapsedTime * 3) * 0.15 + 1.0;
      glowSprite.scale.set(13 * pulse, 13 * pulse, 1);
      centerGlow.intensity = 10 + Math.sin(elapsedTime * 4) * 3;

      // Hacer que todas las flores sigan a la cámara
      sunflowersList.forEach(flower => {
        flower.lookAt(camera.position);
      });
      centerSunflowers.forEach(flower => {
        flower.lookAt(camera.position);
      });

      renderer.render(scene, camera);
    }

    animate();

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });
  </script>
